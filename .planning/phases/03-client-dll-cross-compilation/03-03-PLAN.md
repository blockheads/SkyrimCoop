2. `Code/client/Games/Skyrim/AnimationExperiments.cpp` -- Replace any `#pragma optimize("", off)` and `#pragma optimize("", on)` with:
```cpp
#ifdef _MSC_VER
#pragma optimize("", off)
#else
#pragma GCC optimize("O0")
#endif
```
And at the end:
```cpp
#ifdef _MSC_VER
#pragma optimize("", on)
#endif
```

3. Run first compile attempt:
```bash

xmake build SkyrimTogetherClient -j$(nproc)
```

4. **Iterative fix loop:** For each compilation error:
   - Read the error message carefully
   - Identify the MSVC-ism or missing guard
   - Apply the minimal fix (prefer `#ifdef _MSC_VER` guards over rewrites)
   - Common expected errors and fixes:
     - Missing `#pragma comment(lib, ...)` elsewhere: delete them
     - `strcpy_s` not found: add `#define __STDC_WANT_LIB_EXT1__ 1` before include, or replace with `strncpy` + null terminator
     - Case-sensitive header includes (`Windows.h` vs `windows.h`): lowercase them
     - `__forceinline` errors: add `#ifndef __forceinline` / `#define __forceinline __attribute__((always_inline)) inline` / `#endif` in PCH
     - Missing type `_PIFV`: already guarded in Memory.cpp from Plan 01
     - Link errors for CEF/Discord symbols: means a guard was missed -- go back and add it
   - Re-run `xmake build SkyrimTogetherClient` after each fix
   - Continue until build succeeds with zero errors

5. After successful build, verify the output DLL exists:
```bash
find build/ -name "SkyrimTogetherClient*" -type f
```

Create `tests/smoke_test_client.sh` -- a smoke test that verifies the MinGW-compiled DLL can load into Skyrim and connect to a local server (per D-08).

```bash
#!/bin/bash
# smoke_test_client.sh -- Verify MinGW client DLL loads and connects to server
# Phase 3 BUILD-05 validation
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_ROOT="$(cd "$SCRIPT_DIR/.." && pwd)"

# Configurable paths
DLL_PATH="${DLL_PATH:-$(find "$PROJECT_ROOT/build" -name 'SkyrimTogetherClient.dll' -o -name 'libSkyrimTogetherClient.dll' 2>/dev/null | head -1)}"
SERVER_BIN="${SERVER_BIN:-$(find "$PROJECT_ROOT/build" -name 'SkyrimTogetherServer' -type f 2>/dev/null | head -1)}"
SKYRIM_DIR="${SKYRIM_DIR:-$HOME/.steam/steam/steamapps/common/Skyrim Special Edition}"
SKSE_PLUGINS_DIR="$SKYRIM_DIR/Data/SKSE/Plugins"
LOG_DIR="/tmp/skyrimcoop_smoke_$$"
TIMEOUT="${TIMEOUT:-90}"

# Colors
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

cleanup() {
    echo -e "\n${YELLOW}Cleaning up...${NC}"
    [ -n "${SERVER_PID:-}" ] && kill "$SERVER_PID" 2>/dev/null || true
    # Restore original DLL if we backed it up
    if [ -f "$SKSE_PLUGINS_DIR/SkyrimTogetherClient.dll.bak" ]; then
        mv "$SKSE_PLUGINS_DIR/SkyrimTogetherClient.dll.bak" \
           "$SKSE_PLUGINS_DIR/SkyrimTogetherClient.dll" 2>/dev/null || true
    fi
    rm -rf "$LOG_DIR"
}
trap cleanup EXIT

# Validation
echo "=== SkyrimCoop Smoke Test ==="
echo "DLL:     $DLL_PATH"
echo "Server:  $SERVER_BIN"
echo "Skyrim:  $SKYRIM_DIR"
echo ""

if [ -z "$DLL_PATH" ] || [ ! -f "$DLL_PATH" ]; then
    echo -e "${RED}FAIL: Client DLL not found at $DLL_PATH${NC}"
    echo "Build first: xmake f -p mingw --mingw=/usr && xmake build SkyrimTogetherClient"
    exit 1
fi

if [ ! -d "$SKYRIM_DIR" ]; then
    echo -e "${RED}FAIL: Skyrim directory not found at $SKYRIM_DIR${NC}"
    echo "Set SKYRIM_DIR environment variable to your Skyrim SE installation"
    exit 1
fi

mkdir -p "$LOG_DIR"
mkdir -p "$SKSE_PLUGINS_DIR"

# Step 1: Verify DLL is valid PE file
echo -n "Checking DLL format... "
if file "$DLL_PATH" | grep -q "PE32+"; then
    echo -e "${GREEN}OK${NC} (PE32+ x86-64)"
else
    echo -e "${RED}FAIL${NC} (not a valid PE32+ DLL)"
    file "$DLL_PATH"
    exit 1
fi

# Step 2: Start local server (if available)
if [ -n "$SERVER_BIN" ] && [ -f "$SERVER_BIN" ]; then
    echo -n "Starting local server... "
    "$SERVER_BIN" > "$LOG_DIR/server.log" 2>&1 &
    SERVER_PID=$!
    sleep 3
    if kill -0 "$SERVER_PID" 2>/dev/null; then
        echo -e "${GREEN}OK${NC} (PID: $SERVER_PID)"
    else
        echo -e "${YELLOW}WARN${NC}: Server exited early (may not affect DLL load test)"
        SERVER_PID=""
    fi
else
    echo -e "${YELLOW}SKIP${NC}: Server binary not found, testing DLL load only"
fi

# Step 3: Deploy DLL
echo -n "Deploying DLL to SKSE plugins... "
if [ -f "$SKSE_PLUGINS_DIR/SkyrimTogetherClient.dll" ]; then
    cp "$SKSE_PLUGINS_DIR/SkyrimTogetherClient.dll" \
       "$SKSE_PLUGINS_DIR/SkyrimTogetherClient.dll.bak"
fi
cp "$DLL_PATH" "$SKSE_PLUGINS_DIR/SkyrimTogetherClient.dll"
echo -e "${GREEN}OK${NC}"

# Step 4: Report
echo ""
echo "=== Smoke Test Preparation Complete ==="
echo ""
echo "DLL deployed to: $SKSE_PLUGINS_DIR/SkyrimTogetherClient.dll"
if [ -n "${SERVER_PID:-}" ]; then
    echo "Local server running on PID: $SERVER_PID"
fi
echo ""
echo "Manual verification required:"
echo "  1. Launch Skyrim SE via Proton/Steam"
echo "  2. Check SKSE log for SkyrimTogetherClient plugin load"
echo "  3. If server is running, verify connection in client logs"
echo ""
echo -e "${GREEN}PASS${NC}: DLL is valid PE32+ and deployed successfully"
echo "Full BUILD-05 validation requires manual Proton launch (see above)"
exit 0
```
