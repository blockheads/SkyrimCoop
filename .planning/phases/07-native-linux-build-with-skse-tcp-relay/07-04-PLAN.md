    ```cpp
    // proc_memory.h
    #pragma once
    #include <cstdint>
    #include <cstddef>
    #include <sys/types.h>

    class ProcMemReader {
    public:
        ~ProcMemReader();

        // Open /proc/pid/mem with ptrace SEIZE for Yama bypass
        bool Open(pid_t pid);
        void Close();

        // Read `len` bytes from address `addr` into `buf`
        // Returns bytes read, or -1 on error
        ssize_t Read(void* buf, size_t len, uint64_t addr);

        // Typed read helper
        template<typename T>
        T ReadValue(uint64_t addr) {
            T val{};
            Read(&val, sizeof(T), addr);
            return val;
        }

        bool IsOpen() const { return m_fd != -1; }
        pid_t GetPid() const { return m_pid; }

    private:
        int m_fd = -1;
        pid_t m_pid = 0;
        bool m_isTracer = false;
    };
    ```

    Implementation follows LethalInjection exactly:
    - `Open()`: Check `GetPtraceScope()` by reading `/proc/sys/kernel/yama/ptrace_scope`. If scope >= 1 and pid != getpid(), call `ptrace(PTRACE_SEIZE, pid, nullptr, nullptr)`. Then `open("/proc/{pid}/mem", O_RDONLY)`. Use spdlog for logging (native binary has spdlog).
    - `Read()`: `pread(m_fd, buf, len, static_cast<off_t>(addr))`.
    - `Close()`: Close fd. If tracer, `ptrace(PTRACE_INTERRUPT, pid)`, `waitpid(pid, &status, __WALL)`, `ptrace(PTRACE_DETACH, pid)`. Same pattern as LethalInjection lines 101-127.

    **tcp_client.h / tcp_client.cpp** — Linux TCP client connecting to DLL server.

    ```cpp
    // tcp_client.h
    #pragma once
    #include <cstdint>
    #include "protocol.h"

    class TcpClient {
    public:
        // Connect to localhost:port. Returns true on success.
        bool Connect(uint16_t port);
        void Disconnect();

        // Send a packet (e.g., command to DLL). Returns false on error.
        bool Send(const void* data, uint32_t size);

        // Receive next packet. Blocks until header + payload available.
        // Returns false on disconnect.
        bool Receive(PacketHeader* outHeader, void* outPayload, uint32_t maxPayload);

        // Send a control packet (e.g., CTRL_READY)
        bool SendControl(uint16_t opcode);

        bool IsConnected() const { return m_socket != -1; }

    private:
        int m_socket = -1;

        // Helper: read exactly `len` bytes from socket
        bool ReadExact(void* buf, size_t len);
    };
    ```

    Implementation: standard POSIX socket API (`socket`, `connect`, `send`, `recv`). `ReadExact` loops `recv` until all bytes received or error. Use `127.0.0.1` as host.

    **game_reader.h / game_reader.cpp** — Typed memory reads against game struct offsets. This is the API that rewritten services will use instead of direct pointer dereferences.

    ```cpp
    // game_reader.h
    #pragma once
    #include <cstdint>
    #include <glm/vec3.hpp>
    #include "proc_memory.h"
    #include "pointer_table.h"

    class GameReader {
    public:
        GameReader(ProcMemReader& mem, PointerTable& ptrs)
            : m_mem(mem), m_ptrs(ptrs) {}

        // Read a value at ptr + offset
        template<typename T>
        T Read(uint64_t ptr, uint64_t offset = 0) {
            return m_mem.ReadValue<T>(ptr + offset);
        }

        // Read a block of bytes
        ssize_t ReadBytes(void* buf, size_t len, uint64_t ptr, uint64_t offset = 0) {
            return m_mem.Read(buf, len, ptr + offset);
        }

        // Follow a pointer chain: read ptr at base+offset, then read T at result+finalOffset
        template<typename T>
        T ReadChain(uint64_t base, uint64_t ptrOffset, uint64_t finalOffset) {
            uint64_t intermediate = Read<uint64_t>(base, ptrOffset);
            if (intermediate == 0) return T{};
            return Read<T>(intermediate, finalOffset);
        }

        // Convenience: read actor position (TESObjectREFR::position at offset 0x54)
        glm::vec3 ReadActorPosition(uint64_t actorPtr);

        // Convenience: read form ID (TESForm::formID at offset 0x14)
        uint32_t ReadFormId(uint64_t formPtr);

        // Convenience: read actor health (via ActorValueOwner)
        float ReadActorHealth(uint64_t actorPtr);

        // Lookup by formId, returns 0 if not found
        uint64_t LookupPointer(uint32_t formId) { return m_ptrs.Lookup(formId); }

        ProcMemReader& GetMem() { return m_mem; }
        PointerTable& GetPointers() { return m_ptrs; }

    private:
        ProcMemReader& m_mem;
        PointerTable& m_ptrs;
    };
    ```

    For `ReadActorPosition`: Use offset 0x54 from TESObjectREFR (NiPoint3 position). Read 12 bytes (3 floats), construct glm::vec3.
    For `ReadFormId`: Use offset 0x14 from TESForm.
    For `ReadActorHealth`: This requires following the ActorValueOwner vtable -- stub for now, return 0.0f. Will be filled in during service rewrite.

    Parse args: `--pid <pid>` and `--port <port>`.
    1. Parse command line arguments
    2. Initialize spdlog
    3. Initialize rpmalloc
