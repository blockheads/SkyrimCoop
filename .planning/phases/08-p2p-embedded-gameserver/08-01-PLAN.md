<objective>
Create the LoopbackTransport layer that bridges the native client services to the embedded GameServer without enet6, plus the GameServer modifications needed to support it.

Purpose: This is the core transport mechanism for P2P host mode (per D-04, D-05). The host player's messages flow through in-process loopback instead of network sockets, achieving zero latency while reusing all existing server message handlers unchanged.

Output: LoopbackTransport.h/.cpp, ServerMessageRouter.h/.cpp, modified GameServer.h/.cpp, loopback_transport_tests.cpp
</objective>

    ```cpp
    #pragma once
    #include <cstdint>
    #include <functional>
    #include <TiltedCore/Buffer.hpp>
    #include <Messages/Message.h>

    using TiltedPhoques::ConnectionId_t;
    namespace Server { struct GameServer; }

    class LoopbackTransport {
    public:
        // Reserved ConnectionId for host player -- must not collide with enet6 peer handles
        // enet6 assigns ConnectionId from peer index (small integers). Use a large constant.
        static constexpr ConnectionId_t kHostConnectionId = 0xFFFF'FFFF'0000'0001ULL;

        explicit LoopbackTransport(Server::GameServer& aServer);

        // Serialize a ClientMessage and inject into GameServer::OnConsume
        void SendToServer(const ClientMessage& acMessage);

        // Callback type for receiving ServerMessages from the GameServer
        using ServerMessageCallback = std::function<void(const void*, uint32_t)>;
        void SetServerMessageCallback(ServerMessageCallback aCallback);

        // Called by GameServer::Send when target is kHostConnectionId
        void DeliverToHost(const void* apData, uint32_t aSize);

    private:
        Server::GameServer& m_server;
        ServerMessageCallback m_onServerMessage;
    };
    ```

    **LoopbackTransport.cpp** -- Implement:
    - `SendToServer`: Use the same serialization as MockClient::SendGameMessage:
      1. Create `TiltedPhoques::Buffer buffer(1 << 16)`
      2. Create `Buffer::Writer writer(&buffer)`
      3. `writer.WriteBits(0, 8)` -- first byte reserved for packet type
      4. `acMessage.Serialize(writer)`
      5. Call `m_server.InjectMessage(buffer.GetWriteData() + 1, static_cast<uint32_t>(writer.Size()) - 1, kHostConnectionId)` -- skip first byte because OnConsume expects data AFTER the packet-type byte (Server::HandleMessage strips pData[0] before calling OnConsume)

    ```cpp
    class ServerMessageRouter {
    public:
        // Process raw server message bytes (from loopback callback)
        void OnServerMessage(const void* apData, uint32_t aSize);

        // Opcode-specific callbacks (set by native services)
        using AuthCallback = std::function<void(bool accepted, uint32_t playerId)>;
        using MoveCallback = std::function<void(uint32_t serverId, float x, float y, float z, float rx, float ry)>;

        void SetAuthCallback(AuthCallback aCb);
        void SetMoveCallback(MoveCallback aCb);

        uint32_t GetLastOpcode() const { return m_lastOpcode; }
        bool IsAuthenticated() const { return m_authenticated; }

    private:
        AuthCallback m_onAuth;
        MoveCallback m_onMove;
        uint32_t m_lastOpcode{0};
        bool m_authenticated{false};
    };
    ```
