// How production client sends a ClientMessage over enet6:
bool TransportService::Send(const ClientMessage& acMessage) const noexcept {
    Buffer buffer(1 << 16);
    Buffer::Writer writer(&buffer);
    writer.WriteBits(0, 8); // First byte reserved for packet type
    acMessage.Serialize(writer);
    TiltedPhoques::PacketView packet(reinterpret_cast<char*>(buffer.GetWriteData()), writer.Size());
    Client::Send(&packet);
    return true;
}

**Step 2: Create Code/tests/mock_client.h**

Create a MockClient struct that inherits TiltedPhoques::Client and provides:
- `SendGameMessage(const ClientMessage& acMessage)` -- serializes a ClientMessage into a Packet and sends via enet6 (same pattern as TransportService::Send)
- `OnConsume()` override that uses ServerMessageFactory::Extract to deserialize incoming server messages
- State tracking: `m_connected` bool, `m_lastOpcode` (last received server message opcode), `m_authResponse` (stores the AuthenticationResponse for inspection)
- `OnConnected()` override that sets m_connected = true
- `OnDisconnected()` override that sets m_connected = false
- `OnUpdate()` override (empty)
- Helper: `WaitForConnection(Server::GameServer& server, int maxIterations = 200)` that pumps both server.Update() and this->Update() in a loop with 10ms sleeps until IsConnected() or timeout

```cpp
#pragma once
#include <Client.hpp>
#include <Messages/Message.h>
#include <Messages/AuthenticationResponse.h>
#include <atomic>

namespace Server { struct GameServer; }

struct MockClient : TiltedPhoques::Client
{
    MockClient() noexcept;
    ~MockClient() override;

    void SendGameMessage(const ClientMessage& acMessage);

    // Pump helper: calls server.Update() and this->Update() until connected or timeout
    bool WaitForConnection(Server::GameServer& aServer, int aMaxIterations = 200);

    // Pump helper: calls server.Update() and this->Update() for N iterations
    void Pump(Server::GameServer& aServer, int aIterations = 50);

    bool m_connected{false};
    uint32_t m_lastOpcode{0};
    bool m_authAccepted{false};
    AuthenticationResponse::ResponseType m_authResponseType{};

protected:
    void OnConsume(const void* apData, uint32_t aSize) override;
    void OnConnected() override;
    void OnDisconnected(EDisconnectReason aReason) override;
    void OnUpdate() override;
};
```

  void MockClient::OnConsume(const void* apData, uint32_t aSize) {
      TiltedPhoques::Buffer buf(aSize);
      // Copy incoming data into buffer
      std::memcpy(buf.GetWriteData(), apData, aSize);
      buf.m_size = aSize; // or however Buffer tracks size
      TiltedPhoques::Buffer::Reader reader(&buf);
      // ServerMessageFactory extracts opcode and deserializes
      const ServerMessageFactory factory;
      auto pMessage = factory.Extract(reader);
      if (pMessage) {
          m_lastOpcode = pMessage->GetOpcode();
          if (pMessage->GetOpcode() == AuthenticationResponse::Opcode) {
              auto* pAuth = static_cast<AuthenticationResponse*>(pMessage.get());
              m_authResponseType = pAuth->Type;
              m_authAccepted = (pAuth->Type == AuthenticationResponse::ResponseType::kAccepted);
          }
      }
  }
