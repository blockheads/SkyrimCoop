           local timeout=60
           local elapsed=0
           while [ $elapsed -lt $timeout ]; do
               if [ -f "$relay_info" ]; then
                   # Parse port and pid from JSON (simple grep approach)
                   local port=$(grep -o '"port":[0-9]*' "$relay_info" | grep -o '[0-9]*')
                   local pid=$(grep -o '"pid":[0-9]*' "$relay_info" | grep -o '[0-9]*')

                       echo "$label relay ready: port=$port pid=$pid"
                       echo "$port $pid"
                       return 0
                   fi
               fi
               sleep 1
               elapsed=$((elapsed + 1))
           done
           echo "ERROR: $label relay_info.json not found after ${timeout}s"
           return 1

       echo "=== Phase 10: Two-Player Co-op Launch ==="

       echo "--- Starting CLIENT Skyrim ---"

           echo "Done"
