    **Step 2: Implement DrainCommandQueue in relay_main.cpp**

    Replace the existing stub (lines 118-126) with:
    ```cpp
    static void DrainCommandQueue()
    {
        using TSetPosition = void (__fastcall*)(void*, void*);
        auto pRealSetPosition = reinterpret_cast<TSetPosition>(GetRealSetPosition());

        g_commandQueue.Drain([pRealSetPosition](const CommandSlot& cmd) {
            switch (cmd.opcode) {
            case CMD_SET_POSITION: {
                // args[0] = actorPtr
                // args[1] = (posY_bits << 32) | posX_bits
                // args[2] = (rotX_bits << 32) | posZ_bits
                // args[3] = rotZ_bits (unused for now -- SetPosition only takes NiPoint3)
                void* pActor = reinterpret_cast<void*>(cmd.args[0]);
                if (!pActor || !pRealSetPosition) break;

                uint32_t posXBits = static_cast<uint32_t>(cmd.args[1]);
                uint32_t posYBits = static_cast<uint32_t>(cmd.args[1] >> 32);
                uint32_t posZBits = static_cast<uint32_t>(cmd.args[2]);

                float pos[3];
                memcpy(&pos[0], &posXBits, sizeof(float));
                memcpy(&pos[1], &posYBits, sizeof(float));
                memcpy(&pos[2], &posZBits, sizeof(float));

                // Call the real SetPosition(REFR*, NiPoint3*)
                // NiPoint3 is just 3 floats, so we pass &pos[0]
                pRealSetPosition(pActor, pos);
                break;
            }
            default:
                RelayLog("[SkyrimCoopHooks] Unknown command opcode=0x%04X\n", cmd.opcode);
                break;
            }
        });
    }
    ```

    Remove the `[[maybe_unused]]` attribute from DrainCommandQueue.

    ```cpp
    // Forward declaration (from relay_main.cpp)
    // Already in hook_trampolines.h if we added it there

    static void __fastcall HookActorProcess(void* apThis, float aDelta)
    {
        // Drain command queue once per frame on player actor only (per D-08)
        if (ReadFormId(apThis) == 0x14)
        {

        }

        uint64_t args[3] = { reinterpret_cast<uint64_t>(apThis),

                              FloatToU64(aDelta) };
        SendHookEvent(HOOK_ACTOR_PROCESS, 3, args);
        RealActorProcess(apThis, aDelta);
    }
    ```
