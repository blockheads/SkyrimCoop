    memcpy(&xBits, &x, sizeof(float));
    memcpy(&yBits, &y, sizeof(float));
    memcpy(&zBits, &z, sizeof(float));
    memcpy(&rxBits, &rx, sizeof(float));
    memcpy(&ryBits, &ry, sizeof(float));

    slot.args[1] = (static_cast<uint64_t>(yBits) << 32) | xBits;

**relay_main.cpp -- Add DrainCommandQueue handlers:**

In DrainCommandQueue, add cases before the default:

```cpp
case CMD_SPAWN_ACTOR: {
    uint32_t formId = static_cast<uint32_t>(cmd.args[0]);

    // Unpack position
    uint32_t xBits = static_cast<uint32_t>(cmd.args[1]);
    uint32_t yBits = static_cast<uint32_t>(cmd.args[1] >> 32);
    uint32_t zBits = static_cast<uint32_t>(cmd.args[2]);
    float pos[3];
    memcpy(&pos[0], &xBits, sizeof(float));
    memcpy(&pos[1], &yBits, sizeof(float));
    memcpy(&pos[2], &zBits, sizeof(float));

    // Unpack rotation
    uint32_t rxBits = static_cast<uint32_t>(cmd.args[3]);
    uint32_t ryBits = static_cast<uint32_t>(cmd.args[3] >> 32);
    float rot[3] = {0};
    memcpy(&rot[0], &rxBits, sizeof(float));
    memcpy(&rot[1], &ryBits, sizeof(float));

    // Type aliases for game function signatures
    using TGetFormById = void*(__cdecl*)(uint32_t);
    using THeapAllocate = void*(__fastcall*)(void*, size_t, size_t, bool);
    using TCharCtor = void*(__fastcall*)(void*);
    using TSetLevelMod = void(__fastcall*)(void*, uint32_t);
    // SpawnNewREFR: __fastcall thiscall with 12 params
    // (ModManager*, uint32_t&, void*, float*, float*, void*, void*, void*, uintptr_t, uintptr_t, char, char)
    using TSpawnNewREFR = uint32_t&(__fastcall*)(void*, uint32_t&, void*, float*, float*, void*, void*, void*, uintptr_t, uintptr_t, char, char);

    auto pGetById = reinterpret_cast<TGetFormById>(GetFormById());
    auto pHeapAlloc = reinterpret_cast<THeapAllocate>(GetGameHeapAllocate());
    auto pCharCtor = reinterpret_cast<TCharCtor>(GetCharacterCtor());
    auto pSpawnREFR = reinterpret_cast<TSpawnNewREFR>(GetSpawnNewREFR());
    auto pSetLevelMod = reinterpret_cast<TSetLevelMod>(GetSetLevelMod());
    void* pHeapInst = GetGameHeapInstance() ? *reinterpret_cast<void**>(GetGameHeapInstance()) : nullptr;
    void* pModMgr = GetModManagerInstance() ? *reinterpret_cast<void**>(GetModManagerInstance()) : nullptr;

    if (!pGetById || !pHeapAlloc || !pCharCtor || !pSpawnREFR || !pHeapInst || !pModMgr) {
        RelayLog("[SkyrimCoopHooks] CMD_SPAWN_ACTOR: missing game function pointers\n");
        nlohmann::json result = {
            {JsonField::kOp, JsonOp::kSpawnResult},
            {JsonField::kFormId, formId},
            {JsonField::kPtr, 0},
            {JsonField::kSuccess, false}
        };
        g_tcpServer.SendJson(result);
        break;
    }

    // 1. Look up base NPC form
    void* pNpc = pGetById(formId);
    if (!pNpc) {
        RelayLog("[SkyrimCoopHooks] CMD_SPAWN_ACTOR: GetById(%u) returned null\n", formId);
        nlohmann::json result = {
            {JsonField::kOp, JsonOp::kSpawnResult},
            {JsonField::kFormId, formId},
            {JsonField::kPtr, 0},
            {JsonField::kSuccess, false}
        };
        g_tcpServer.SendJson(result);
        break;
    }

    // 2. Get player for cell/worldspace context
    void* pPlayer = pGetById(0x14);
    if (!pPlayer) {
        RelayLog("[SkyrimCoopHooks] CMD_SPAWN_ACTOR: player (0x14) not found\n");
        nlohmann::json result = {
            {JsonField::kOp, JsonOp::kSpawnResult},
            {JsonField::kFormId, formId},
            {JsonField::kPtr, 0},
            {JsonField::kSuccess, false}
        };
        g_tcpServer.SendJson(result);
        break;
    }

    // Read parentCell from player (offset varies, but TESObjectREFR::parentCell is at 0x70 in SE 1.6.x)
    // and worldspace at 0x78
    void* pCell = *reinterpret_cast<void**>(reinterpret_cast<uintptr_t>(pPlayer) + 0x70);
    void* pWorldSpace = *reinterpret_cast<void**>(reinterpret_cast<uintptr_t>(pPlayer) + 0x78);

    // 3. Allocate Actor via game heap + character constructor
    // STR Actor::New: Memory::Allocate<Actor>() -> CharacterConstructor(pActor)
    // Actor size is approximately 0x2B0 in SE 1.6.x (from ExActor struct)
    void* pActor = pHeapAlloc(pHeapInst, 0x2B0, 0, false);
    if (!pActor) {
        RelayLog("[SkyrimCoopHooks] CMD_SPAWN_ACTOR: heap alloc failed\n");
        nlohmann::json result = {
            {JsonField::kOp, JsonOp::kSpawnResult},
            {JsonField::kFormId, formId},
            {JsonField::kPtr, 0},
            {JsonField::kSuccess, false}
        };
        g_tcpServer.SendJson(result);
        break;
    }
    memset(pActor, 0, 0x2B0);
    pCharCtor(pActor);

    // 4. SetSkipSaveFlag: unk10 = 0xFFFF (offset 0x10 from TESForm base)
    *reinterpret_cast<uint16_t*>(reinterpret_cast<uintptr_t>(pActor) + 0x10) = 0xFFFF;

    // 5. SetLevelMod(pActor->extraData, 4) -- extraData offset is 0x70 in TESObjectREFR base
    // Actually, Actor.extraData is the BSExtraDataList, offset from start of Actor
    // In STR: pActor->extraData which is ExtraDataList, SetLevelMod is called on &extraData
    // For simplicity, pass extraData pointer (actor + 0x70 is parentCell, extraData is at different offset)
    // Actually looking at STR Actor::SetLevelMod: `const auto pExtraDataList = &extraData;`
    // TESObjectREFR has extraData at offset 0x70 in older versions... but we need to be careful.
    // The safest approach: skip SetLevelMod for Phase 15 MVP (actor will still spawn without level scaling)
    // We can add it in a future fix if needed.

    // 6. SetParentCell: virtual call at vtable index -- skip for now, SpawnNewREFR handles cell
    // 7. SetBaseForm: baseForm pointer at offset 0x20 from TESForm
    *reinterpret_cast<void**>(reinterpret_cast<uintptr_t>(pActor) + 0x20) = pNpc;

    // 8. For SpawnNewREFR: if cell is interior, pass it; if exterior, pass nullptr (per STR pattern)
    // Check cell flag: cellFlags[0] & 1 means interior
    void* pCellForSpawn = pCell;
    if (pCell) {
        // cell->cellFlags is at offset 0x44 (approximate for SE 1.6.x)
        uint8_t cellFlags = *reinterpret_cast<uint8_t*>(reinterpret_cast<uintptr_t>(pCell) + 0x44);
        if (!(cellFlags & 1)) {
            pCellForSpawn = nullptr; // exterior cell -- pass nullptr per STR
        }
    }

    // 9. SpawnNewREFR(modManager, refrHandle, baseForm, &pos, &rot, cell, worldspace, actor, 0, 0, 0, 1)
    uint32_t refrHandle = 0;
    pSpawnREFR(pModMgr, refrHandle, pNpc, pos, rot, pCellForSpawn, pWorldSpace, pActor, 0, 0, 0, 1);

    // 10. ForcePosition using existing SetPosition
    if (pRealSetPosition && pActor) {
        pRealSetPosition(pActor, pos);
    }

    RelayLog("[SkyrimCoopHooks] CMD_SPAWN_ACTOR: spawned formId=%u ptr=%p\n", formId, pActor);

    nlohmann::json result = {
        {JsonField::kOp, JsonOp::kSpawnResult},
        {JsonField::kFormId, formId},
        {JsonField::kPtr, reinterpret_cast<uint64_t>(pActor)},
        {JsonField::kSuccess, true}
    };
    g_tcpServer.SendJson(result);
    break;
}

case CMD_REMOVE_ACTOR: {
    void* pActor = reinterpret_cast<void*>(cmd.args[0]);
    if (!pActor) {
        RelayLog("[SkyrimCoopHooks] CMD_REMOVE_ACTOR: null actor pointer\n");
        break;
    }

    // Per D-06: Immediate deletion.
    // The simplest safe approach: disable the actor by setting its flags.
    // SetSkipSaveFlag (unk10 = 0xFFFF) already set on spawn.
    // For actual deletion, we need to call TESObjectREFR::Delete or similar.
    // STR uses `actor->Disable()` + `actor->SetDelete(true)` pattern which sets flags
    // that the engine will clean up.
    //
    // Disable: Set flag bit 11 (0x800) on flags at offset 0x0C (TESForm::flags)
    // SetDelete: Set flag bit 5 (0x20) on flags at offset 0x0C
    uint32_t& flags = *reinterpret_cast<uint32_t*>(reinterpret_cast<uintptr_t>(pActor) + 0x0C);
    flags |= (1 << 11); // kDisabled
    flags |= (1 << 5);  // kMarkedForDeletion
    // Also mark changed to make engine process the deletion
    // MarkChanged(0x1) at offset 0x0C -> already handled by flag set

    RelayLog("[SkyrimCoopHooks] CMD_REMOVE_ACTOR: marked for deletion ptr=%p\n", pActor);
    break;
}
```
