    **CharacterService.cpp changes:**
    - Replace stub OnHookEvent with full implementation. Current code switches on `static_cast<HookOpcode>(aPkt.header.opcode)`. New code:
      ```cpp
      void CharacterService::OnHookEvent(const nlohmann::json& aMsg) {
          const auto& op = aMsg["op"].get_ref<const std::string&>();
          if (op == JsonOp::kActorAdded) {
              HandleActorAdded(aMsg["ptr"].get<uint64_t>(), aMsg["form_id"].get<uint32_t>());
          } else if (op == JsonOp::kActorRemoved) {
              HandleActorRemoved(aMsg["ptr"].get<uint64_t>(), aMsg["form_id"].get<uint32_t>());
          } else if (op == JsonOp::kSetPosition) {
              uint32_t formId = aMsg["form_id"].get<uint32_t>();
              if (formId != 0) {
                  float px = aMsg["x"].get<float>();
                  float py = aMsg["y"].get<float>();
                  float pz = aMsg["z"].get<float>();
                  float rx = aMsg["rx"].get<float>();
                  float rz = aMsg["rz"].get<float>();
                  m_positionCache[formId] = {glm::vec3(px, py, pz), glm::vec2(rx, rz)};
              }
          } else if (op == JsonOp::kCharacterCtor) {
              HandleCharacterCtor(aMsg["ptr"].get<uint64_t>(), aMsg["form_id"].get<uint32_t>());
          } else if (op == JsonOp::kSpawnActor) {
              HandleSpawnActor(aMsg["ptr"].get<uint64_t>(), aMsg["form_id"].get<uint32_t>());
          } else if (op == JsonOp::kDeathItems) {
              HandleDeathItems(aMsg["ptr"].get<uint64_t>(), aMsg["form_id"].get<uint32_t>());
          } else if (op == JsonOp::kPerformAction) {
              HandlePerformAction(aMsg["actor_form_id"].get<uint32_t>(), aMsg["action_form_id"].get<uint32_t>());
          } else if (op == JsonOp::kActorProcess) {
              // Pointer table handled in main.cpp already
          } else if (op == JsonOp::kPlayAnimation) {
              // Handle animation event
          } else if (op == JsonOp::kMountPackage) {
              // Handle mount package event
          }
      }
      ```
    - Replace command senders (SendSetPosition, SendPlayAnimation, etc.) to build JSON + m_client.SendJsonLine():
      ```cpp
      void CharacterService::SendSetPosition(uint64_t aActorPtr, const glm::vec3& aPos, const glm::vec2& aRot) {
          nlohmann::json cmd = {
              {"op", JsonOp::kCmdSetPosition},
              {"actor_ptr", aActorPtr},
              {"x", aPos.x}, {"y", aPos.y}, {"z", aPos.z}
          };
          m_client.SendJsonLine(cmd);
      }
      ```
    - Remove all `FloatFromU64` / `U64ToFloat` usage -- JSON gives native floats.
    - Remove any remaining `#include "protocol.h"` if present (json_protocol.h already included from Plan 03 stub).
    - If CharacterService.h needs any additional member declarations for new command sender signatures, update accordingly.

    **Key pattern:**
    - Replace `aPkt.args[N]` with `aMsg["field_name"].get<type>()`
    - Replace `FloatFromU64(aPkt.args[N])` with `aMsg["field_name"].get<float>()` -- no conversion needed
    - Replace CommandPacket construction with nlohmann::json object + m_client.SendJsonLine()
