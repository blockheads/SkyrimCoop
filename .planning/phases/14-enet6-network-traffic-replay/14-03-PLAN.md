#include <catch2/catch.hpp>
#include "server_replay_harness.h"
#include "Recording/MessageSerializer.h"
#include "Recording/Base64.h"
#include <Messages/AuthenticationRequest.h>
#include <Messages/AuthenticationResponse.h>
#include <nlohmann/json.hpp>
#include <fstream>
#include <string>

static std::string EnetFixturePath(const char* aName) {
    return std::string(FIXTURE_PATH) + "/enet/" + aName;
}

// Generate fixture from known message data
static void GenerateAuthFixture() {
    // Create AuthenticationRequest with known token
    AuthenticationRequest req;
    req.Token = "test_token_abc123";
    // Serialize to JSON
    auto j = MessageSerializer::ClientToJson(req);
    j["dir"] = "c2s";
    j["peer"] = 1;
    j["ts"] = 0.0;
    // Write to fixture file
    std::ofstream out(EnetFixturePath("auth_flow.ndjson"));
    out << j.dump() << "\n";
}

TEST_CASE("Generate auth fixture", "[Enet6][Auth][!hide]") {
    GenerateAuthFixture();
}

TEST_CASE("Auth replay creates player in server", "[Enet6][Auth]") {
    // Ensure fixture exists
    GenerateAuthFixture();

    ServerReplayHarness harness;
    size_t count = harness.ReplayFile(EnetFixturePath("auth_flow.ndjson"), 1);

    REQUIRE(count == 1);

    // Server should have processed the auth and created a Player
    auto& playerManager = harness.GetWorld().GetPlayerManager();
    // After auth, GetByConnectionId(1) should return a valid Player
    auto* pPlayer = playerManager.GetByConnectionId(1);
    // Note: auth may reject if token validation fails in test mode.
    // Check captured server messages for an AuthenticationResponse
    auto& captured = harness.GetCapturedServerMessages();
    REQUIRE(captured.size() >= 1);
    CHECK(captured[0]["type"] == "AuthenticationResponse");
    CHECK(captured[0].contains("opcode"));
    CHECK(captured[0]["opcode"] == 0); // kAuthenticationResponse
}
```

#pragma once
#include "replay/replay_harness.h"
#include "server_replay_harness.h"
#include <string>

// DualReplayHarness: coordinator wrapping both client-side (hook event) and
// server-side (enet6 message) replay. Each side replays independently from
// its own recorded NDJSON file (D-06).
class DualReplayHarness {
public:
    DualReplayHarness() = default;

    // Client-side: replay NDJSON hook events through native_client services
    size_t ReplayClientFile(const std::string& aPath);

    // Server-side: replay NDJSON enet6 messages through server services
    size_t ReplayServerFile(const std::string& aPath, TiltedPhoques::ConnectionId_t aPeer = 1);

    // Access for assertions
    ReplayHarness& Client() { return m_client; }
    ServerReplayHarness& Server() { return m_server; }

private:
    ReplayHarness m_client;
    ServerReplayHarness m_server;
};
```
