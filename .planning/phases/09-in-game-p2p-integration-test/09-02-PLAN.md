```cpp
// ---------------------------------------------------------------------------
// JoinClient: enet6 client for join mode (connects to remote host GameServer)
// ---------------------------------------------------------------------------
class JoinClient : public TiltedPhoques::Client
{
public:
    JoinClient(entt::registry& aRegistry, entt::dispatcher& aDispatcher)
        : m_registry(aRegistry)
        , m_dispatcher(aDispatcher)
    {}

    void SendGameMessage(const ClientMessage& acMessage)
    {
        TiltedPhoques::ScopedAllocator _{};
        TiltedPhoques::Buffer buffer(1 << 16);
        TiltedPhoques::Buffer::Writer writer(&buffer);
        writer.WriteBits(0, 8); // packet type byte (0 = game message)
        acMessage.Serialize(writer);
        Send(writer.GetBuffer().data(), writer.Size());
    }

    bool m_connected{false};
    bool m_authAccepted{false};
    bool m_gotAuthResponse{false};

protected:
    void OnConsume(const void* apData, uint32_t aSize) override
    {
        // Skip first byte (packet type), same as MockClient
        if (aSize < 2) return;
        auto* pData = static_cast<const uint8_t*>(apData);

        ClientMessageFactory factory;
        TiltedPhoques::ViewBuffer buf(const_cast<uint8_t*>(pData + 1), aSize - 1);
        TiltedPhoques::Buffer::Reader reader(&buf);

        auto pMessage = factory.Extract(reader);
        if (!pMessage) {
            spdlog::warn("JoinClient: failed to extract server message");
            return;
        }

        // Check for auth response
        if (pMessage->GetOpcode() == AuthenticationResponse::Opcode) {
            auto* pAuth = static_cast<AuthenticationResponse*>(pMessage.get());
            m_gotAuthResponse = true;
            m_authAccepted = (pAuth->Type == AuthenticationResponse::ResponseType::kAccepted);
            spdlog::info("JoinClient: auth response type={}", static_cast<int>(pAuth->Type));
        }

        // Log all received opcodes for debugging
        spdlog::debug("JoinClient: received opcode {}", pMessage->GetOpcode());
    }

    void OnConnected() override
    {
        m_connected = true;
        spdlog::info("JoinClient: connected to host GameServer");
    }

    void OnDisconnected(EDisconnectReason aReason) override
    {
        m_connected = false;
        spdlog::warn("JoinClient: disconnected (reason={})", static_cast<int>(aReason));
    }

    void OnUpdate() override {}

private:
    entt::registry& m_registry;
    entt::dispatcher& m_dispatcher;
};
```

NOTE: The factory extraction uses `ClientMessageFactory` to read server-originated messages -- but server messages use `ServerMessageFactory`. Check mock_client.cpp for the correct pattern. If MockClient uses `ServerMessageFactory`, use that instead. The key is: the server sends `ServerMessage` types, so the join client must deserialize with `ServerMessageFactory`. Update the include to add `#include <Messages/ServerMessageFactory.h>` if not already present, and use `ServerMessageFactory` in OnConsume.
