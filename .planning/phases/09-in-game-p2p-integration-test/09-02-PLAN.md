```cpp
// mock_skyrim_loader.cpp -- Loads relay DLL without Skyrim for headless testing
// Exploits the fact that SKSEPlugin_Load ignores its argument and hooks fail gracefully
#include <windows.h>
#include <winsock2.h>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include "protocol.h"

typedef bool (*SKSEPlugin_Load_t)(void*);

static bool SendHookEvent(SOCKET sock, uint16_t opcode, uint8_t argCount, const uint64_t* args) {
    HookEventPacket pkt{};
    pkt.header.opcode = opcode;
    pkt.header.length = sizeof(pkt) - sizeof(PacketHeader);
    pkt.argCount = argCount;
    for (int i = 0; i < argCount && i < 8; i++)
        pkt.args[i] = args[i];
    return send(sock, (const char*)&pkt, sizeof(pkt), 0) == sizeof(pkt);
}

int main(int argc, char* argv[]) {
    // 1. Parse args: --dll <path> [--events N] [--delay-ms N]
    const char* dllPath = nullptr;
    int eventCount = 10;
    int delayMs = 100;
    for (int i = 1; i < argc; i++) {
        if (strcmp(argv[i], "--dll") == 0 && i+1 < argc) dllPath = argv[++i];
        else if (strcmp(argv[i], "--events") == 0 && i+1 < argc) eventCount = atoi(argv[++i]);
        else if (strcmp(argv[i], "--delay-ms") == 0 && i+1 < argc) delayMs = atoi(argv[++i]);
    }
    if (!dllPath) { fprintf(stderr, "Usage: mock_skyrim_loader --dll <path.dll> [--events N] [--delay-ms N]\n"); return 1; }

    // 2. Initialize winsock
    WSADATA wsa;
    WSAStartup(MAKEWORD(2,2), &wsa);

    // 3. LoadLibrary the relay DLL
    HMODULE hDll = LoadLibraryA(dllPath);
    if (!hDll) { fprintf(stderr, "LoadLibrary failed: %lu\n", GetLastError()); return 1; }

    // 4. Call SKSEPlugin_Load(nullptr) — hooks fail but TCP server starts
    auto pfnLoad = (SKSEPlugin_Load_t)GetProcAddress(hDll, "SKSEPlugin_Load");
    if (!pfnLoad) { fprintf(stderr, "GetProcAddress SKSEPlugin_Load failed\n"); return 1; }
    if (!pfnLoad(nullptr)) { fprintf(stderr, "SKSEPlugin_Load returned false\n"); return 1; }
    printf("DLL loaded, TCP server should be running\n");

    // 5. Read relay_info.json from DLL directory to get TCP port
    //    The DLL writes it next to itself. Parse the port.
    //    Wait up to 5 seconds for it to appear.
    // ... (read relay_info.json, extract port, connect TCP)

    // 6. Connect to DLL TCP server as a mock "native process"
    // ... (TCP connect to 127.0.0.1:port)

    // 7. Send CTRL_READY
    // ... (send ControlPacket with CTRL_READY opcode)

    // 8. Send scripted HookEventPackets (HOOK_ACTOR_ADDED, HOOK_SET_POSITION, etc.)
    //    Loop eventCount times with delayMs between each
    // ... (call SendHookEvent with various opcodes)

    // 9. Send CTRL_SHUTDOWN, cleanup, exit 0
    // ...

    FreeLibrary(hDll);
    WSACleanup();
    printf("mock_skyrim_loader: PASS\n");
    return 0;
}
```
