<objective>
Fix /proc/pid/mem game memory reads so hook event pointers produce valid formId values.

Purpose: All HOOK_CHARACTER_CTOR and HOOK_SPAWN_ACTOR events currently fail with "could not read formId at ptr=..." because pread() returns zero or fails. The verification report (09-VERIFICATION.md) claims this is a Wine guest-to-host address translation issue, but analysis of Wine internals shows PE modules are loaded directly into the Linux process address space (same virtual addresses). The actual root causes are likely: (a) pread() not being tested at all against a known-good address, (b) timing -- hooks fire during object construction before formID is initialized at offset 0x14, (c) possible offset mismatch for the Skyrim version. This plan adds diagnostic probing, deferred reads, and a startup smoke test to identify and fix the real issue.

Output: proc_memory.cpp with FindModuleBase + hex dump helpers, game_reader.cpp with deferred formId read, main.cpp with startup pread smoke test
</objective>

```cpp
uint64_t ProcMemReader::FindModuleBase(const char* aNameSuffix) const {
    std::string suffix(aNameSuffix);
    std::transform(suffix.begin(), suffix.end(), suffix.begin(), ::tolower);
    for (const auto& r : m_regions) {
        if (!r.readable) continue;
        std::string nameLower = r.name;
        std::transform(nameLower.begin(), nameLower.end(), nameLower.begin(), ::tolower);
        if (nameLower.size() >= suffix.size() &&
            nameLower.compare(nameLower.size() - suffix.size(), suffix.size(), suffix) == 0) {
            return r.start;
        }
    }
    return 0;
}
```

```cpp
ssize_t ProcMemReader::HexDump(uint64_t addr, size_t len, const char* aLabel) {
    if (len > 64) len = 64;
    uint8_t buf[64]{};
    ssize_t bytesRead = ::pread(m_fd, buf, len, static_cast<off_t>(addr));
    if (bytesRead <= 0) {
        spdlog::error("HexDump '{}' at {:#x}: pread failed: {} ({})",
            aLabel, addr, strerror(errno), errno);
        return bytesRead;
    }
    std::string hex;
    for (ssize_t i = 0; i < bytesRead; i++) {
        char tmp[4];
        snprintf(tmp, sizeof(tmp), "%02x ", buf[i]);
        hex += tmp;
    }
    spdlog::info("HexDump '{}' at {:#x} ({} bytes): {}", aLabel, addr, bytesRead, hex);
    return bytesRead;
}
```

```cpp
uint32_t GameReader::ReadFormId(uint64_t aFormPtr) {
    uint32_t formId = m_mem.ReadValue<uint32_t>(aFormPtr + cFormIdOffset);
    if (formId == 0) {
        std::string region = m_mem.ValidateAddress(aFormPtr);
        if (region.empty()) {
            spdlog::error("ReadFormId: ptr {:#x} is NOT in any mapped region", aFormPtr);

    }
    return formId;
}
```

**2. Add ReadFormIdDeferred to GameReader (game_reader.h + game_reader.cpp):**

This method retries the formId read after a short delay, working around the race condition where CharacterCtor hooks fire before the formID field is initialized.

In game_reader.h, add to the public section:
```cpp
// Read formId with retry for partially constructed objects.
// Tries up to `aMaxRetries` times with `aDelayUs` microsecond delays.
// Returns 0 if all retries fail.
uint32_t ReadFormIdDeferred(uint64_t aFormPtr, int aMaxRetries = 3, int aDelayUs = 1000);
```

In game_reader.cpp, add:
```cpp
uint32_t GameReader::ReadFormIdDeferred(uint64_t aFormPtr, int aMaxRetries, int aDelayUs) {
    for (int attempt = 0; attempt <= aMaxRetries; attempt++) {
        uint32_t formId = m_mem.ReadValue<uint32_t>(aFormPtr + cFormIdOffset);
        if (formId != 0) {
            if (attempt > 0) {
                spdlog::info("ReadFormIdDeferred: got formId {:#x} at ptr {:#x} on attempt {}",
                    formId, aFormPtr, attempt + 1);
            }
            return formId;
        }
        if (attempt < aMaxRetries) {
            usleep(aDelayUs);
        }
    }
    // All retries failed -- hex dump for diagnostics
    std::string region = m_mem.ValidateAddress(aFormPtr);
    spdlog::warn("ReadFormIdDeferred: ptr {:#x} formId still 0 after {} retries (region: '{}')",
        aFormPtr, aMaxRetries + 1, region.empty() ? "UNMAPPED" : region);
    m_mem.HexDump(aFormPtr, 0x20, "ReadFormIdDeferred exhausted");
    return 0;
}
```

Add `#include <unistd.h>` at the top of game_reader.cpp for usleep().
