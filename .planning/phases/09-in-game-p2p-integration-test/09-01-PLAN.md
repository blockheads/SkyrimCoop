```cpp
#ifdef __linux__
#include <fstream>
#include <sstream>

// Find the base address of SkyrimSE.exe in the target process's memory map
static uint64_t FindSkyrimBaseAddress(pid_t aPid)
{
    std::string mapsPath = "/proc/" + std::to_string(aPid) + "/maps";
    std::ifstream maps(mapsPath);
    if (!maps.is_open()) {
        spdlog::error("verify: cannot open {}", mapsPath);
        return 0;
    }

    std::string line;
    while (std::getline(maps, line)) {
        if (line.find("SkyrimSE.exe") != std::string::npos) {
            // Format: "address-address perms offset dev inode pathname"
            uint64_t baseAddr = 0;
            std::istringstream iss(line);
            std::string addrRange;
            iss >> addrRange;
            // Parse "start-end" format
            auto dashPos = addrRange.find('-');
            if (dashPos != std::string::npos) {
                baseAddr = std::stoull(addrRange.substr(0, dashPos), nullptr, 16);
            }
            spdlog::info("verify: SkyrimSE.exe base address = {:#x}", baseAddr);
            return baseAddr;
        }
    }
    spdlog::error("verify: SkyrimSE.exe not found in /proc/{}/maps", aPid);
    return 0;
}

static bool VerifyHostMemory(pid_t aPid)
{
    spdlog::info("verify: opening /proc/{}/mem for host memory verification", aPid);

    ProcMemReader reader;
    if (!reader.Open(aPid)) {
        spdlog::error("verify: failed to open /proc/{}/mem (ptrace SEIZE failed?)", aPid);
        return false;
    }
    spdlog::info("verify: /proc/{}/mem opened successfully", aPid);

    // Find SkyrimSE.exe base address from /proc/pid/maps
    uint64_t baseAddr = FindSkyrimBaseAddress(aPid);
    if (baseAddr == 0) {
        reader.Close();
        return false;
    }

    // Read the DOS header signature (first 2 bytes of the PE)
    // Should be 'MZ' = 0x5A4D
    uint16_t dosSignature = 0;
    ssize_t bytesRead = reader.Read(&dosSignature, sizeof(dosSignature), baseAddr);
    if (bytesRead != sizeof(dosSignature)) {
        spdlog::error("verify: failed to read DOS header at {:#x} (read {} bytes)",
                      baseAddr, bytesRead);
        reader.Close();
        return false;
    }

    if (dosSignature == 0x5A4D) {
        spdlog::info("verify: DOS header signature OK (MZ = {:#x}) at {:#x}",
                      dosSignature, baseAddr);
    } else {
        spdlog::error("verify: unexpected DOS header {:#x} at {:#x} (expected 0x5A4D)",
                      dosSignature, baseAddr);
        reader.Close();
        return false;
    }

    // Read the PE header offset (at DOS header offset 0x3C)
    uint32_t peOffset = 0;
    reader.Read(&peOffset, sizeof(peOffset), baseAddr + 0x3C);
    spdlog::info("verify: PE header offset = {:#x}", peOffset);

    // Read PE signature (should be "PE\0\0" = 0x00004550)
    uint32_t peSignature = 0;
    reader.Read(&peSignature, sizeof(peSignature), baseAddr + peOffset);
    if (peSignature == 0x00004550) {
        spdlog::info("verify: PE signature OK (PE\\0\\0) at {:#x}", baseAddr + peOffset);
    } else {
        spdlog::error("verify: unexpected PE signature {:#x} (expected 0x00004550)", peSignature);
        reader.Close();
        return false;
    }

    spdlog::info("verify: /proc/pid/mem verification PASSED -- Skyrim process memory is readable");
    reader.Close();
    return true;
}
#endif // __linux__
```
