inject_windowed_ini() {
    local ini_path="$1"
    # Backup
    cp "$ini_path" "${ini_path}.two_player_backup"

    # Remove existing display settings to avoid duplicates
    sed -i '/^bFull Screen=/d; /^iSize H=/d; /^iSize W=/d; /^bAlwaysActive=/d; /^bBorderless=/d; /^sStartingCell=/d' "$ini_path"

    # Inject windowed mode + mouse unlock settings after [General]
    sed -i '/^\[General\]/a sStartingCell=Riverwood' "$ini_path"

    # Inject display settings after [Display]
    sed -i '/^\[Display\]/a bFull Screen=0\niSize H=720\niSize W=1280\nbBorderless=0' "$ini_path"

    # Inject interface settings after [Interface] (create section if needed)
    if ! grep -q '^\[Interface\]' "$ini_path"; then
        echo -e "\n[Interface]" >> "$ini_path"
    fi
    sed -i '/^\[Interface\]/a bAlwaysActive=1' "$ini_path"
}

**Window positioning** (per D-10):
```bash
position_windows() {
    info "Positioning Skyrim windows side-by-side..."
    sleep 5  # Wait for windows to appear

    # Try wmctrl first, fall back to xdotool
    if command -v wmctrl >/dev/null 2>&1; then
        local windows=$(wmctrl -l | grep -i "skyrim" | awk '{print $1}')
        local count=0
        for wid in $windows; do
            if [ $count -eq 0 ]; then
                wmctrl -i -r "$wid" -e '0,0,0,1280,720'
            else
                wmctrl -i -r "$wid" -e '0,1280,0,1280,720'
            fi
            count=$((count + 1))
        done
        if [ $count -ge 2 ]; then
            pass "Positioned $count Skyrim windows side-by-side"
            return 0
        fi
    fi

    if command -v xdotool >/dev/null 2>&1; then
        local windows=$(xdotool search --name "Skyrim" 2>/dev/null || true)
        local count=0
        for wid in $windows; do
            if [ $count -eq 0 ]; then
                xdotool windowmove "$wid" 0 0
                xdotool windowsize "$wid" 1280 720
            else
                xdotool windowmove "$wid" 1280 0
                xdotool windowsize "$wid" 1280 720
            fi
            count=$((count + 1))
        done
        if [ $count -ge 2 ]; then
            pass "Positioned $count Skyrim windows (xdotool)"
            return 0
        fi
    fi

    warn "Could not position windows automatically (install wmctrl or xdotool)"
}
```

**Main flow** implementing the sequential launch from RESEARCH.md:

1. Kill any existing Skyrim/native processes
2. Inject windowed INI for host prefix
3. Launch host Skyrim via Proton
4. Wait for relay_info.json, capture host's TCP port + PID
5. Launch host's `skyrim-coop --pid $HOST_PID --port $HOST_RELAY_PORT`
6. Wait for GameServer on UDP 10578 (reuse ss polling)
7. Clone Proton prefix for join instance
8. Inject windowed INI for join prefix
9. Launch join Skyrim via Proton with join prefix env vars
10. Wait for NEW relay_info.json (different port/PID from host's)
11. Launch join's `skyrim-coop --pid $JOIN_PID --port $JOIN_RELAY_PORT --connect 127.0.0.1:10578`
12. Position windows side-by-side
13. Print "Both instances running. Press Enter to shut down."
14. Wait for user input, then cleanup
